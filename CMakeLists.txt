cmake_minimum_required(VERSION 3.21)

project(BadConnectionStudio
    VERSION 0.1.0
    DESCRIPTION "Degrade live camera/microphone feeds through a real low-bitrate codec round-trip"
    LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_AUTOMOC ON)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)

if(NOT CMAKE_BUILD_TYPE AND NOT CMAKE_CONFIGURATION_TYPES)
    set(CMAKE_BUILD_TYPE RelWithDebInfo CACHE STRING "" FORCE)
endif()

option(BCS_BUILD_TESTS "Build unit / round-trip tests" ON)
option(BCS_BUILD_TOOLS "Build the codec round-trip benchmark tool" ON)

include(GNUInstallDirs)

find_package(Qt6 6.4 REQUIRED COMPONENTS Core Gui Widgets Multimedia)
if(UNIX AND NOT APPLE)
    # Used only for logind sleep/resume notifications; optional.
    find_package(Qt6 6.4 QUIET COMPONENTS DBus)
endif()

find_package(PkgConfig REQUIRED)
pkg_check_modules(FFMPEG REQUIRED IMPORTED_TARGET
    libavcodec libavformat libavdevice libavutil libswscale libswresample)

set(BCS_LINUX OFF)
set(BCS_WINDOWS OFF)
if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
    set(BCS_LINUX ON)
    # PipeWire is loaded at runtime with dlopen() so the AppImage never bundles
    # a client library that mismatches the host's PipeWire modules. Only the
    # headers are needed at build time.
    pkg_check_modules(PIPEWIRE_HEADERS QUIET libpipewire-0.3)
    option(BCS_WITH_PIPEWIRE "Build the native PipeWire virtual-microphone backend" ${PIPEWIRE_HEADERS_FOUND})
elseif(WIN32)
    set(BCS_WINDOWS ON)
endif()

add_subdirectory(src)

if(BCS_BUILD_TOOLS)
    add_subdirectory(tools)
endif()

if(BCS_BUILD_TESTS)
    enable_testing()
    add_subdirectory(tests)
endif()

# ---------------------------------------------------------------- packaging
if(BCS_LINUX)
    install(FILES packaging/linux/bad-connection-studio.desktop
            DESTINATION ${CMAKE_INSTALL_DATADIR}/applications)
    install(FILES resources/bad-connection-studio.svg
            DESTINATION ${CMAKE_INSTALL_DATADIR}/icons/hicolor/scalable/apps)
    install(FILES packaging/linux/io.github.badconnectionstudio.metainfo.xml
            DESTINATION ${CMAKE_INSTALL_DATADIR}/metainfo)
endif()
install(FILES docs/LINUX_SETUP.md docs/WINDOWS_SETUP.md docs/LICENSES.md docs/TESTING.md README.md
        DESTINATION ${CMAKE_INSTALL_DOCDIR})
# Generated by scripts/license-report-*.sh during packaging.
install(FILES ${CMAKE_BINARY_DIR}/LICENSE-REPORT.txt DESTINATION ${CMAKE_INSTALL_DOCDIR} OPTIONAL)

if(BCS_WINDOWS)
    include(packaging/windows/CPackWindows.cmake)
endif()
